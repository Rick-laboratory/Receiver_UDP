cmake_minimum_required(VERSION 3.16)
project(UdpReceiver LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This Winsock example requires Windows.")
endif()
add_library(udp_receiver Receiver_UDP.cpp)
target_include_directories(udp_receiver PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(udp_receiver PUBLIC cxx_std_17)
target_compile_definitions(udp_receiver PUBLIC _WIN32_WINNT=0x0600)
target_link_libraries(udp_receiver PUBLIC ws2_32)
add_executable(udp-receive examples/receive.cpp)
target_link_libraries(udp-receive PRIVATE udp_receiver)
include(CTest)
if(BUILD_TESTING)
    add_executable(loopback-test tests/loopback_test.cpp)
    target_link_libraries(loopback-test PRIVATE udp_receiver)
    add_test(NAME loopback-receiver COMMAND loopback-test)
    set_tests_properties(loopback-receiver PROPERTIES TIMEOUT 15)
endif()
foreach(target udp_receiver udp-receive loopback-test)
    if(TARGET ${target})
        set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /permissive-)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endif()
endforeach()
